   366→// Compute a fare quote for a provider given distance/duration
   367→export function computeFare(
   368→  provider: Provider,
   369→  distanceKm: number,
   370→  durationMin: number,
   371→  surge = 1
   372→) {
   373→  const price =
   374→    (provider.baseFare + provider.perKm * distanceKm + provider.perMin * durationMin) *
   375→    surge;
   376→  return Math.round(price * 100) / 100;
   377→}

   379→// Haversine distance in km
   380→export function haversine(
   381→  a: { lat: number; lng: number },
   382→  b: { lat: number; lng: number }
   383→) {
   384→  const R = 6371;
   385→  const dLat = ((b.lat - a.lat) * Math.PI) / 180;
   386→  const dLng = ((b.lng - a.lng) * Math.PI) / 180;
   387→  const lat1 = (a.lat * Math.PI) / 180;
   388→  const lat2 = (b.lat * Math.PI) / 180;
   389→  const h =
   390→    Math.sin(dLat / 2) ** 2 +
   391→    Math.cos(lat1) * Math.cos(lat2) * Math.sin(dLng / 2) ** 2;
   392→  return 2 * R * Math.asin(Math.sqrt(h));
   393→}
